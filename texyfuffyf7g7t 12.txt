-- ==================== HÀM HỖ TRỢ ====================
local function getCompassDirection(deg)
    local dirs = {"N (Bắc)", "NE (Đông Bắc)", "E (Đông)", "SE (Đông Nam)", "S (Nam)", "SW (Tây Nam)", "W (Tây)", "NW (Tây Bắc)"}
    local index = math.floor((deg + 22.5) / 45) % 8 + 1
    return dirs[index]
end

local function getObjectInCrosshair()
    local cam = Workspace.CurrentCamera
    if not cam then return nil end
    local ray = cam:ViewportPointToRay(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    if LocalPlayer.Character then
        params.FilterDescendantsInstances = {LocalPlayer.Character}
    end
    local result = Workspace:Raycast(ray.Origin, ray.Direction * 500, params)
    if result and result.Instance then
        local hit = result.Instance
        if hit.Parent and hit.Parent:FindFirstChildOfClass("Humanoid") then
            return hit.Parent:FindFirstChild("HumanoidRootPart") or hit
        elseif not hit.Anchored then
            return hit
        end
    end
    return nil
end

local function restoreVisibility()
    for _, part in ipairs(hiddenParts) do
        if part and part.Parent then part.LocalTransparencyModifier = 0 end
    end
    hiddenParts = {}
end

local function hardResetCamera()
    isSteering = false
    HUD.Visible = false
    pcall(function() UserInputService.MouseBehavior = Enum.MouseBehavior.Default end)
    
    if renderConn then renderConn:Disconnect() renderConn = nil end
    if touchConn then touchConn:Disconnect() touchConn = nil end
    if inputConn then inputConn:Disconnect() inputConn = nil end

    restoreVisibility()

    local cam = Workspace.CurrentCamera
    local char = LocalPlayer.Character
    if cam and char then
        cam.CameraType = Enum.CameraType.Custom
        local hum = char:FindFirstChildOfClass("Humanoid")
        local hrp = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Head")
        if hum then cam.CameraSubject = hum end
        if hrp then cam.CFrame = CFrame.new(hrp.Position - (hrp.CFrame.LookVector * 12) + Vector3.new(0, 3, 0), hrp.Position) end
    end

    LocalPlayer.MinCameraZoomDistance = 0.5
    LocalPlayer.MaxCameraZoomDistance = 128

    DriveBtn.Text = "2. LÁI FPV: TẮT"
    DriveBtn.BackgroundColor3 = Color3.fromRGB(231, 76, 60)
end

local function startSteering()
    if not targetPart or not targetPart.Parent then return hardResetCamera() end
    
    isSteering = true
    HUD.Visible = true
    DriveBtn.Text = "2. LÁI FPV: BẬT"
    DriveBtn.BackgroundColor3 = Color3.fromRGB(46, 204, 113)

    batteryPercent = 100
    batteryVoltage = 16.8

    local cam = Workspace.CurrentCamera
    local currentLook = cam.CFrame.LookVector
    cameraYaw = math.deg(math.atan2(-currentLook.X, -currentLook.Z))
    cameraPitch = math.deg(math.asin(currentLook.Y))
    cam.CameraType = Enum.CameraType.Scriptable
    
    restoreVisibility()
    local rootModel = targetPart:FindFirstAncestorOfClass("Model") or targetPart.Parent
    if rootModel then
        for _, descendant in ipairs(rootModel:GetDescendants()) do
            if descendant:IsA("BasePart") then table.insert(hiddenParts, descendant) end
        end
    elseif targetPart:IsA("BasePart") then
        table.insert(hiddenParts, targetPart)
    end

    local size = targetPart.Size
    local frontOffset = (size.Z / 2) + 1.2
    local topOffset = (size.Y / 2) + 0.8

    if inputConn then inputConn:Disconnect() end
    inputConn = UserInputService.InputChanged:Connect(function(input)
        if isSteering and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Delta
            cameraYaw = cameraYaw - (delta.X * sensitivity)
            cameraPitch = math.clamp(cameraPitch - (delta.Y * sensitivity), -80, 80)
        end
    end)

    if renderConn then renderConn:Disconnect() end
    renderConn = RunService.RenderStepped:Connect(function(dt)
        if not isSteering or not targetPart or not targetPart.Parent or not targetPart:IsDescendantOf(Workspace) then
            hardResetCamera() return
        end

        UserInputService.MouseBehavior = Enum.MouseBehavior.LockCenter
        local rotCFrame = CFrame.Angles(0, math.rad(cameraYaw), 0) * CFrame.Angles(math.rad(cameraPitch), 0, 0)
        targetPart.Anchored = false
        targetPart.CFrame = CFrame.new(targetPart.Position) * rotCFrame
        targetPart.AssemblyLinearVelocity = rotCFrame.LookVector * flySpeed
        cam.CFrame = targetPart.CFrame * CFrame.new(0, topOffset, -frontOffset)
        
        for _, part in ipairs(hiddenParts) do
            if part and part.Parent then part.LocalTransparencyModifier = 1 end
        end

        local pos = targetPart.Position
        local myChar = LocalPlayer.Character
        local charPos = myChar and myChar:FindFirstChild("HumanoidRootPart") and myChar.HumanoidRootPart.Position or pos
        local dist = (pos - charPos).Magnitude

        local signalPercent = math.clamp(math.floor(100 - (dist / maxSignalRange * 100)), 0, 100)
        local rssi = math.clamp(math.floor(-35 - (dist * 0.012) + math.random(-2, 2)), -100, -30)

        batteryPercent = math.max(0, batteryPercent - (dt * 0.4))
        batteryVoltage = 13.5 + (3.3 * (batteryPercent / 100))
        currentDraw = 12.0 + math.random(-5, 8) / 10 + (flySpeed / 20)
        local watts = math.floor(batteryVoltage * currentDraw)

        if batteryPercent <= 0 or signalPercent <= 0 then hardResetCamera() return end

        local noiseIntensity = (100 - signalPercent) / 100
        if math.random(1, 2) == 1 then
            ScanLine1.Position = UDim2.new(0, 0, math.random(), 0)
            ScanLine1.Size = UDim2.new(1, 0, 0, math.random(2, 6 + math.floor(noiseIntensity * 10)))
            ScanLine1.BackgroundTransparency = math.clamp(0.9 - (noiseIntensity * 0.5), 0.3, 0.9)
            ScanLine2.Position = UDim2.new(0, 0, math.random(), 0)
            ScanLine2.Size = UDim2.new(1, 0, 0, math.random(1, 4 + math.floor(noiseIntensity * 8)))
            ScanLine2.BackgroundTransparency = math.clamp(0.95 - (noiseIntensity * 0.5), 0.4, 0.95)
        end
        NoiseOverlay.BackgroundTransparency = math.clamp(0.97 - (noiseIntensity * 0.3), 0.6, 0.97)

        local activeBars = signalPercent >= 75 and 4 or signalPercent >= 50 and 3 or signalPercent >= 25 and 2 or signalPercent >= 10 and 1 or 0
        local themeColor = signalPercent <= 20 and Color3.fromRGB(231, 76, 60) or Color3.fromRGB(0, 255, 127)

        for i = 1, 4 do
            if i <= activeBars then
                signalBars[i].BackgroundColor3 = themeColor
                signalBars[i].BackgroundTransparency = 0
            else
                signalBars[i].BackgroundColor3 = Color3.fromRGB(90, 90, 90)
                signalBars[i].BackgroundTransparency = 0.5
            end
        end

        RSSILabel.Text = string.format("%d dBm", rssi)
        RSSILabel.TextColor3 = themeColor
        PercentLabel.Text = string.format("%d%%", signalPercent)
        PercentLabel.TextColor3 = themeColor

        local heading = math.floor((360 - cameraYaw) % 360)
        CompassLabel.Text = string.format("%03d° %s", heading, getCompassDirection(heading))

        BatteryLabel.Text = string.format("BAT: %d%%\nVOLT: %.1fV\nPWR: %.1fA / %dW", math.floor(batteryPercent), batteryVoltage, currentDraw, watts)
        BatteryLabel.TextColor3 = batteryPercent <= 15 and Color3.fromRGB(231, 76, 60) or Color3.fromRGB(0, 255, 127)

        TelemetryLabel.Text = string.format("POS: X:%d | Y:%d | Z:%d  |  GÓC NGHIÊNG: %d°", math.floor(pos.X), math.floor(pos.Y), math.floor(pos.Z), math.floor(cameraPitch))
    end)

    if touchConn then touchConn:Disconnect() end
    touchConn = targetPart.Touched:Connect(function(hit)
        if not isSteering or not targetPart then return end
        if hit:IsDescendantOf(LocalPlayer.Character) or hit == targetPart then return end
        if touchConn then touchConn:Disconnect() touchConn = nil end
        pcall(function() targetPart.AssemblyLinearVelocity = Vector3.zero end)
        task.defer(function() hardResetCamera() end)
    end)
end

-- ==================== CẬP NHẬT DANH SÁCH NGƯỜI CHƠI TRONG SERVER ====================
local function updatePlayerListUI()
    for _, child in ipairs(ScrollingPlayers:GetChildren()) do
        if child:IsA("TextButton") then child:Destroy() end
    end
    
    local count = 0
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            count = count + 1
            local pBtn = Instance.new("TextButton", ScrollingPlayers)
            pBtn.Size = UDim2.new(1, 0, 0, 26)
            pBtn.Font = Enum.Font.SourceSansBold
            pBtn.TextSize = 12
            pBtn.Text = p.Name
            
            local isSelected = false
            for _, target in ipairs(selectedTargets) do
                if target.player == p then isSelected = true break end
            end
            
            if isSelected then
                pBtn.BackgroundColor3 = Color3.fromRGB(46, 204, 113)
                pBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
            else
                pBtn.BackgroundColor3 = Color3.fromRGB(45, 45, 55)
                pBtn.TextColor3 = Color3.fromRGB(200, 200, 200)
            end
            
            Instance.new("UICorner", pBtn).CornerRadius = UDim.new(0, 4)
            
            pBtn.MouseButton1Click:Connect(function()
                local char = p.Character
                local hrp = char and char:FindFirstChild("HumanoidRootPart")
                if hrp then
                    local foundIdx = nil
                    for idx, target in ipairs(selectedTargets) do
                        if target.player == p then foundIdx = idx break end
                    end
                    
                    if foundIdx then
                        table.remove(selectedTargets, foundIdx)
                        pBtn.BackgroundColor3 = Color3.fromRGB(45, 45, 55)
                        pBtn.TextColor3 = Color3.fromRGB(200, 200, 200)
                    else
                        table.insert(selectedTargets, {player = p, hrp = hrp})
                        pBtn.BackgroundColor3 = Color3.fromRGB(46, 204, 113)
                        pBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
                    end
                end
            end)
        end
    end
    ScrollingPlayers.CanvasSize = UDim2.new(0, 0, 0, count * 30)
end

            bg.AlwaysOnTop = true
            local dot = Instance.new("Frame", bg)
            dot.Size = UDim2.new(1, 0, 1, 0)
            dot.BackgroundColor3 = Color3.fromRGB(255, 0, 0)
            Instance.new("UICorner", dot).CornerRadius = UDim.new(1, 0)
            
            table.insert(b52Markers, bg)
            table.insert(dotsList, dot)
            
            local hl = Instance.new("Highlight", CoreGui)
            hl.Adornee = model
            hl.FillColor = Color3.fromRGB(255, 0, 0)
            hl.OutlineColor = Color3.fromRGB(255, 0, 0)
            hl.FillTransparency = 0.4
            table.insert(b52Markers, hl)
        end
    end
    
    if blinkConn then blinkConn:Disconnect() end
    blinkConn = RunService.RenderStepped:Connect(function()
        local alpha = (math.sin(tick() * 15) + 1) / 2
        for _, dot in ipairs(dotsList) do
            if dot and dot.Parent then
                dot.BackgroundTransparency = alpha * 0.8
            end
        end
    end)
    
    -- Khởi tạo danh sách bom đang bay và gán vị trí xuất phát cao hơn mục tiêu 600 khối
    local activeBombs = {}
    if #targetsHRP > 0 and #selectedBombs > 0 then
        for _, bomb in ipairs(selectedBombs) do
            if bomb and bomb.Parent then
                local randomVictim = targetsHRP[math.random(1, #targetsHRP)]
                if randomVictim and randomVictim.Parent then
                    bomb.Anchored = false
                    local dropPos = randomVictim.Position + Vector3.new(0, 600, 0)
                    bomb.CFrame = CFrame.new(dropPos)
                    table.insert(activeBombs, {bomb = bomb, target = randomVictim})
                end
            end
        end
    end
    
    -- Vòng lặp cập nhật liên tục: Tên lửa bám theo vị trí ngang của mục tiêu di chuyển nhưng không làm thay đổi tốc độ rớt xuống (-200)
    if b52MoveConn then b52MoveConn:Disconnect() end
    b52MoveConn = RunService.Heartbeat:Connect(function()
        if not b52Active then return end
        for _, item in ipairs(activeBombs) do
            local bomb = item.bomb
            local target = item.target
            if bomb and bomb.Parent and target and target.Parent then
                local targetPos = target.Position
                local currentPos = bomb.Position
                -- Tính hướng bám theo phương ngang (X, Z) của người chơi
                local dir = Vector3.new(targetPos.X - currentPos.X, 0, targetPos.Z - currentPos.Z)
                if dir.Magnitude > 2 then
                    dir = dir.Unit
                    -- Giữ nguyên vận tốc rơi xuống theo chiều dọc là -200, phần ngang điều chỉnh hướng bám theo tốc độ 150
                    bomb.AssemblyLinearVelocity = Vector3.new(dir.X * 150, -200, dir.Z * 150)
                else
                    bomb.AssemblyLinearVelocity = Vector3.new(0, -200, 0)
                end
            end
        end
    end)
    
    task.delay(5, function()
        b52Active = false
        if blinkConn then blinkConn:Disconnect() blinkConn = nil end
        if b52MoveConn then b52MoveConn:Disconnect() b52MoveConn = nil end
        B52Btn.Text = "KÍCH HOẠT B52 🚀"
        B52Btn.BackgroundColor3 = Color3.fromRGB(150, 0, 0)
        B52Btn.TextColor3 = Color3.fromRGB(255, 255, 0)
        for _, m in ipairs(b52Markers) do m:Destroy() end
        b52Markers = {}
        dotsList = {}
        hardResetCamera()
    end)
end)
