local function getEggInfo(eggName)
    if EggPetDatabase[eggName] then return EggPetDatabase[eggName] end
    for key, data in pairs(EggPetDatabase) do
        if eggName:lower():find(key:lower()) or key:lower():find(eggName:lower()) then
            return data
        end
    end
    return {Pet = "???", Income = "???"}
end

local function getRarity(name)
    local l = name:lower()
    if l:find("divine") or l:find("nightflame") or l:find("kitsune") then return "Divine", 10 end
    if l:find("eternal") or l:find("gorilla king") or l:find("ice dragon") or l:find("lava dragon") then return "Eternal", 9 end
    if l:find("secret") or l:find("cerberus") or l:find("kraken") or l:find("king snake") then return "Secret", 8 end
    if l:find("cosmic") or l:find("unicorn") or l:find("leviathan") or l:find("royal sphinx") then return "Cosmic", 7 end
    if l:find("mythic") or l:find("spider") or l:find("mammoth") then return "Mythic", 6 end
    return "Common", 1
end

-- === Hàm chuyển tab ===
local function switchTab(tabBtn)
    for btn, container in pairs(TabContainers) do
        container.Visible = false
        TweenService:Create(btn, TweenInfo.new(0.2), {
            BackgroundColor3 = Color3.fromRGB(40, 40, 60),
            TextColor3 = Color3.fromRGB(180, 180, 200)
        }):Play()
    end
    TabContainers[tabBtn].Visible = true
    TweenService:Create(tabBtn, TweenInfo.new(0.2), {
        BackgroundColor3 = Color3.fromRGB(255, 100, 150),
        TextColor3 = Color3.fromRGB(255, 255, 255)
    }):Play()
    CurrentTab = tabBtn
end

                            pcall(function() r:FireServer(best) end)
                        end
                    end
                end
                pcall(function()
                    firetouchinterest(root, best, 0)
                    task.wait(0.05)
                    firetouchinterest(root, best, 1)
                end)
            end
        end
    end
end)

        if Config.AutoRare and getRoot() then

            for _, obj in pairs(Workspace:GetDescendants()) do
                if (obj:IsA("BasePart") or obj:IsA("Model")) and obj.Name:lower():find("egg") and not obj.Name:lower():find("spawn") then
                    local _, prio = getRarity(obj.Name)
                    local minP = RarityPriority[Config.MinRarity] or 8

                        local p = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
                        if p then
                            local d = (p.Position - root.Position).Magnitude
                            if prio > bestP or (prio == bestP and d < bestD) then
                                bestP = prio; bestD = d; best = p
                            end
                        end

end)
