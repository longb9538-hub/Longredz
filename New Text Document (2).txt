local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")
local TweenService = game:GetService("TweenService")

-- // VARIABLES
local LocalPlayer = Players.LocalPlayer
local Camera = workspace.CurrentCamera
local FOV_ACTIVE = false
local DEFAULT_FOV = 70
local TARGET_FOV = 120
local currentFOV = DEFAULT_FOV
local isMobile = false
local menuVisible = false

-- // KIỂM TRA NỀN TẢNG
isMobile = UserInputService.TouchEnabled and not UserInputService.KeyboardEnabled and not UserInputService.MouseEnabled

-- // CHỜ PLAYER SPAWN
if not LocalPlayer.Character or not LocalPlayer.Character:FindFirstChild("Humanoid") then
    LocalPlayer.CharacterAdded:Wait()
end

-- // ĐẢM BẢO CAMERA TỒN TẠI
repeat task.wait() until Camera

-- // ==================== FOV FUNCTIONS ====================
local function SetFOV(targetFOV, speed)
    speed = speed or 0.15
    local tweenInfo = TweenInfo.new(speed, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
    local tween = TweenService:Create(Camera, tweenInfo, {FieldOfView = targetFOV})
    tween:Play()
    currentFOV = targetFOV
end

local function Toggle120FOV()
    FOV_ACTIVE = not FOV_ACTIVE
    
    if FOV_ACTIVE then
        SetFOV(TARGET_FOV, 0.2)
    else
        SetFOV(DEFAULT_FOV, 0.15)
    end
end

local function DynamicFOV()
    local humanoid = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid")
    
    if humanoid and FOV_ACTIVE then
        local velocity = humanoid.MoveDirection.Magnitude
        local dynamicFOV = TARGET_FOV + (velocity * 5)
        dynamicFOV = math.clamp(dynamicFOV, TARGET_FOV, 130)
        
        if math.abs(Camera.FieldOfView - dynamicFOV) > 0.5 then
            Camera.FieldOfView = Camera.FieldOfView:Lerp(dynamicFOV, 0.1)
        end
    end
end

RunService.RenderStepped:Connect(function()
    if FOV_ACTIVE then
        DynamicFOV()
    end
end)

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    
    -- F4 để toggle FOV
    if input.KeyCode == Enum.KeyCode.F4 then
        Toggle120FOV()
        if MenuObjects then
            MenuObjects.UpdateStatusUI()
        end
    end
    
    -- RightControl để toggle menu
    if input.KeyCode == Enum.KeyCode.RightControl then
        if MenuObjects then
            if menuVisible then
                MenuObjects.HideMenu()
            else
                MenuObjects.ShowMenu()
            end
        end
    end
end)
