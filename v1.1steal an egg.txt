local function getRoot()
    local char = LocalPlayer.Character
    return char and char:FindFirstChild("HumanoidRootPart")
end

local function getHumanoid()
    local char = LocalPlayer.Character
    return char and char:FindFirstChildOfClass("Humanoid")
end

local function getEggInfo(eggName)
    if EggPetDatabase[eggName] then return EggPetDatabase[eggName] end
    for key, data in pairs(EggPetDatabase) do
        if eggName:lower():find(key:lower()) or key:lower():find(eggName:lower()) then
            return data
        end
    end
    return {Pet = "???", Income = "???"}
end

local function getRarity(name)
    local l = name:lower()
    if l:find("divine") or l:find("nightflame") or l:find("kitsune") then return "Divine", 10 end
    if l:find("eternal") or l:find("gorilla king") or l:find("ice dragon") or l:find("lava dragon") then return "Eternal", 9 end
    if l:find("secret") or l:find("cerberus") or l:find("kraken") or l:find("king snake") then return "Secret", 8 end
    if l:find("cosmic") or l:find("unicorn") or l:find("leviathan") or l:find("royal sphinx") then return "Cosmic", 7 end
    if l:find("mythic") or l:find("spider") or l:find("mammoth") then return "Mythic", 6 end
    return "Common", 1
end

local function findMyBase()
    for _, obj in pairs(Workspace:GetChildren()) do
        if obj:IsA("Model") or obj:IsA("Folder") then
            if obj.Name:lower():find(LocalPlayer.Name:lower()) then
                local base = obj:FindFirstChild("Base") or obj:FindFirstChild("Spawn") or obj.PrimaryPart
                if base then return base end
            end
        end
    end
    for _, obj in pairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") and obj.Name:lower():find("base") then
            local owner = obj:GetAttribute("Owner") or obj:GetAttribute("Player")
            if owner == LocalPlayer.Name then return obj end
        end
    end
    for _, obj in pairs(Workspace:GetChildren()) do
        if obj:IsA("SpawnLocation") then
            local owner = obj:GetAttribute("Owner") or obj:GetAttribute("Player")
            if owner == LocalPlayer.Name then return obj end
        end
    end
    return Workspace:FindFirstChildOfClass("SpawnLocation")
end

local function lockEgg(eggPart)
    if not eggPart or not Config.LockEgg then return end
    if eggPart:IsA("BasePart") then
        pcall(function()
            eggPart.Anchored = true
            eggPart.CanCollide = false
            eggPart.Massless = true
            eggPart.CanTouch = false
            eggPart.CanQuery = false
        end)
    end
    CarriedEgg = eggPart
end

local function unlockEgg(eggPart)
    eggPart = eggPart or CarriedEgg
    if not eggPart then return end
    if eggPart:IsA("BasePart") then
        pcall(function()
            eggPart.Anchored = false
            eggPart.CanCollide = true
            eggPart.Massless = false
            eggPart.CanTouch = true
            eggPart.CanQuery = true
        end)
    end
    CarriedEgg = nil
end

local function startCarryLock(eggPart)
    if CarryConnection then CarryConnection:Disconnect() end
    if not eggPart then return end
    CarryConnection = RunService.Heartbeat:Connect(function()
        local root = getRoot()
        if not root or not eggPart or not eggPart.Parent then return end
        if not CarriedEgg or CarriedEgg ~= eggPart then return end
        local holdPos = root.CFrame * CFrame.new(0, 0, -Config.HoldDistance)
        eggPart.CFrame = holdPos
        pcall(function()
            eggPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            eggPart.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
        end)
    end)
end

local function stopCarryLock()
    if CarryConnection then
        CarryConnection:Disconnect()
        CarryConnection = nil
    end
end

-- 🚀 HỆ THỐNG BAY MƯỢT
-- ══════════════════════════════════════════════════════════
local function startFly()
    local root = getRoot()
    if not root then return end
    if flyBV then flyBV:Destroy() end
    if flyBG then flyBG:Destroy() end
    flyBV = Instance.new("BodyVelocity")
    flyBV.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    flyBV.Velocity = Vector3.new(0, 0, 0)
    flyBV.P = 1250
    flyBV.Parent = root
    flyBG = Instance.new("BodyGyro")
    flyBG.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    flyBG.P = 10000
    flyBG.D = 100
    flyBG.CFrame = root.CFrame
    flyBG.Parent = root
end

local function stopFly()
    if flyBV then flyBV:Destroy() flyBV = nil end
    if flyBG then flyBG:Destroy() flyBG = nil end
end

local function flyTo(targetPos, timeout)
    timeout = timeout or 15
    local root = getRoot()
    if not root then return false end
    startFly()
    local startTime = tick()
    while tick() - startTime < timeout do
        if not flyBV or not flyBV.Parent then break end
        local curRoot = getRoot()
        if not curRoot then break end
        local delta = targetPos - curRoot.Position
        local dist = delta.Magnitude
        if dist < 3 then
            flyBV.Velocity = Vector3.new(0, 0, 0)
            break
        end
        local dir = delta.Unit
        flyBV.Velocity = dir * Config.FlySpeed
        flyBG.CFrame = CFrame.new(curRoot.Position, curRoot.Position + dir)
        RunService.Heartbeat:Wait()
    end
    stopFly()
    return true
end

-- ══════════════════════════════════════════════════════════
-- 🎯 TÌM TRỨNG
-- ══════════════════════════════════════════════════════════
local function findBestEgg()
    local root = getRoot()
    if not root then return nil end
    local best, bestPrio, bestDist = nil, 0, math.huge
    local minPrio = RarityPriority[Config.MinRarity] or 8
    for _, obj in pairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") or obj:IsA("Model") then
            local name = obj.Name
            if name:lower():find("egg") 
                and not name:lower():find("spawn") 
                and not name:lower():find("nest")
                and not name:lower():find("base")
                and obj ~= CarriedEgg then
                local _, prio = getRarity(name)
                if prio >= minPrio then
                    local p = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
                    if p then
                        local d = (p.Position - root.Position).Magnitude
                        if prio > bestPrio or (prio == bestPrio and d < bestDist) then
                            bestPrio = prio
                            bestDist = d
                            best = {Part = p, Name = name, Rarity = getRarity(name), Priority = prio}
                        end
                    end
                end
            end
        end
    end
    return best
end

local function findNearestEgg()
    local root = getRoot()
    if not root then return nil end
    local best, bestD = nil, math.huge
    for _, obj in pairs(Workspace:GetDescendants()) do
        if (obj:IsA("BasePart") or obj:IsA("Model")) and obj.Name:lower():find("egg") and not obj.Name:lower():find("spawn") and obj ~= CarriedEgg then
            local p = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
            if p then
                local d = (p.Position - root.Position).Magnitude
                if d < bestD then bestD = d; best = p end
            end
        end
    end
    return best
end

local function fireStealRemotes(eggPart)
    for _, r in pairs(ReplicatedStorage:GetDescendants()) do
        if r:IsA("RemoteEvent") then
            local n = r.Name:lower()
            if n:find("steal") or n:find("take") or n:find("collect") or n:find("grab") then
                pcall(function() r:FireServer(eggPart) end)
            end
        end
    end
end

local function fireDepositRemotes(eggPart)
    for _, r in pairs(ReplicatedStorage:GetDescendants()) do
        if r:IsA("RemoteEvent") or r:IsA("RemoteFunction") then
            local n = r.Name:lower()
            if n:find("deposit") or n:find("place") or n:find("put") or n:find("store") then
                pcall(function()
                    if r:IsA("RemoteEvent") then r:FireServer(eggPart) else r:InvokeServer(eggPart) end
                end)
            end
        end
    end
end

local function stealAndCarry(eggData)
    local root = getRoot()
    if not root or not eggData then return false end
    
    -- Bay đến
    flyTo(eggData.Part.Position + Vector3.new(0, 3, 0), 15)
    task.wait(0.2)
    
    -- Steal
    fireStealRemotes(eggData.Part)
    pcall(function()
        firetouchinterest(root, eggData.Part, 0)
        task.wait(0.1)
        firetouchinterest(root, eggData.Part, 1)
    end)
    task.wait(0.5)
    
    -- 🔒 Khóa trứng
    if Config.LockEgg then
        lockEgg(eggData.Part)
        startCarryLock(eggData.Part)
    end
    task.wait(1.5)
    
    -- Bay về base (vẫn khóa)
    local base = findMyBase()
    if base then
        flyTo(base.Position + Vector3.new(0, 5, 0), 25)
        task.wait(0.5)
        if Config.AutoDeposit then
            fireDepositRemotes(eggData.Part)
        end
        task.wait(0.5)
    end
    
    -- Mở khóa
    stopCarryLock()
    task.wait(0.3)
    unlockEgg(eggData.Part)
    return true
end

local function createToggle(container, text, emoji, color, default, callback)
    local Btn = Instance.new("TextButton")
    Btn.Size = UDim2.new(1, -10, 0, 48)
    Btn.BackgroundColor3 = Color3.fromRGB(30, 30, 48)
    Btn.Text = ""

    BCorner.Parent = Btn

    local BStroke = Instance.new("UIStroke")
    BStroke.Color = Color3.fromRGB(60, 60, 90)
    BStroke.Thickness = 1
    BStroke.Parent = Btn

    local Icon = Instance.new("TextLabel")
    Icon.Size = UDim2.new(0, 45, 1, 0)
    Icon.BackgroundTransparency = 1
    Icon.Text = emoji
    Icon.TextSize = 22

    Label.Position = UDim2.new(0, 50, 0, 0)
    Label.BackgroundTransparency = 1
    Label.Text = text
    Label.TextColor3 = Color3.fromRGB(230, 230, 240)
    Label.TextSize = 14
    Label.Font = Enum.Font.GothamBold

    Label.Parent = Btn

    local Ind = Instance.new("Frame")
    Ind.Size = UDim2.new(0, 50, 0, 24)
    Ind.Position = UDim2.new(1, -60, 0.5, -12)
    Ind.BackgroundColor3 = Color3.fromRGB(50, 50, 70)

    Ind.Parent = Btn

    local ICorner = Instance.new("UICorner")

    local Dot = Instance.new("Frame")

    Dot.Position = UDim2.new(0, 3, 0.5, -9)
    Dot.BackgroundColor3 = Color3.fromRGB(150, 150, 170)
    Dot.BorderSizePixel = 0
    Dot.Parent = Ind

    local DCorner = Instance.new("UICorner")
    DCorner.CornerRadius = UDim.new(1, 0)
    DCorner.Parent = Dot

    local state = default or false
    if state then
        Ind.BackgroundColor3 = color
        Dot.Position = UDim2.new(1, -21, 0.5, -9)
        Dot.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
        BStroke.Color = color
        callback(true)
    end

    Btn.MouseButton1Click:Connect(function()
        state = not state
        callback(state)
        if state then
            TweenService:Create(Ind, TweenInfo.new(0.25), {BackgroundColor3 = color}):Play()
            TweenService:Create(Dot, TweenInfo.new(0.25, Enum.EasingStyle.Back), {
                Position = UDim2.new(1, -21, 0.5, -9),
                BackgroundColor3 = Color3.fromRGB(255, 255, 255)
            }):Play()
            TweenService:Create(BStroke, TweenInfo.new(0.25), {Color = color}):Play()
        else
            TweenService:Create(Ind, TweenInfo.new(0.25), {BackgroundColor3 = Color3.fromRGB(50, 50, 70)}):Play()
            TweenService:Create(Dot, TweenInfo.new(0.25, Enum.EasingStyle.Back), {
                Position = UDim2.new(0, 3, 0.5, -9),
                BackgroundColor3 = Color3.fromRGB(150, 150, 170)
            }):Play()
            TweenService:Create(BStroke, TweenInfo.new(0.25), {Color = Color3.fromRGB(60, 60, 90)}):Play()
        end
    end)

    return Btn
end
