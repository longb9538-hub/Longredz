local function getCompassDirection(deg)
    local dirs = {"N (Bắc)", "NE (Đông Bắc)", "E (Đông)", "SE (Đông Nam)", "S (Nam)", "SW (Tây Nam)", "W (Tây)", "NW (Tây Bắc)"}
    local index = math.floor((deg + 22.5) / 45) % 8 + 1
    return dirs[index]
end

local function getObjectInCrosshair()
    local cam = Workspace.CurrentCamera
    if not cam then return nil end
    local ray = cam:ViewportPointToRay(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    if LocalPlayer.Character then
        params.FilterDescendantsInstances = {LocalPlayer.Character}
    end
    local result = Workspace:Raycast(ray.Origin, ray.Direction * 500, params)
    if result and result.Instance then
        local hit = result.Instance
        if hit.Parent and hit.Parent:FindFirstChildOfClass("Humanoid") then
            return hit.Parent:FindFirstChild("HumanoidRootPart") or hit
        elseif not hit.Anchored then
            return hit
        end
    end
    return nil
end

-- ==================== DRONE THÁM HIỂM KẼ HỞ & NGÓC NGÁCH ====================
local function scanMapForSecrets(dronePos)
    for _, hl in ipairs(droneHighlights) do
        if hl then hl:Destroy() end
    end
    droneHighlights = {}

    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            local name = obj.Name:lower()
            local isSecretOrEntrance = name:find("door") or name:find("secret") or name:find("base") or name:find("hidden") or name:find("entrance") or name:find("tunnel") or name:find("pipe") or (obj.Transparency > 0 and obj.Transparency < 1)
            
            local dist = (obj.Position - dronePos).Magnitude
            if dist < 800 then 
                table.insert(scannedMapData.general, obj.Position)

                if isSecretOrEntrance then
                    -- Tránh thêm trùng lặp quá gần
                    local duplicate = false
                    for _, pos in ipairs(scannedMapData.entrances) do
                        if (pos - obj.Position).Magnitude < 15 then duplicate = true break end
                    end
                    
                    if not duplicate then
                        table.insert(scannedMapData.entrances, obj.Position)
                    end

                    local hl = Instance.new("Highlight", CoreGui)
                    hl.Adornee = obj.Parent:IsA("Model") and obj.Parent or obj
                    hl.FillColor = Color3.fromRGB(46, 204, 113)
                    hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                    hl.FillTransparency = 0.3
                    table.insert(droneHighlights, hl)
                end
            end
        end
    end
end

local function startDroneSteering()
    if not selectedDrone or not selectedDrone.Parent then return hardResetCamera() end
    
    isDroneSteering = true
    isDroneReturning = false
    isPenetratingGap = false
    currentEntranceTarget = nil
    droneTimer = 0
    droneStartPosition = selectedDrone.Position
    
    scannedMapData = { blocked = {}, entrances = {}, general = {} }
    
    HUD.Visible = true
    DroneDriveBtn.Text = "4. DRONE THÁM HIỂM KẼ HỞ: BẬT"
    DroneDriveBtn.BackgroundColor3 = Color3.fromRGB(0, 180, 216)

    batteryPercent = 100
    batteryVoltage = 16.8

    local cam = Workspace.CurrentCamera
    cam.CameraType = Enum.CameraType.Scriptable
    
    restoreDroneVisibility()
    local rootModel = selectedDrone:FindFirstAncestorOfClass("Model") or selectedDrone.Parent
    if rootModel then
        for _, descendant in ipairs(rootModel:GetDescendants()) do
            if descendant:IsA("BasePart") then table.insert(droneHiddenParts, descendant) end
        end
    elseif selectedDrone:IsA("BasePart") then
        table.insert(droneHiddenParts, selectedDrone)
    end

    local size = selectedDrone.Size
    local frontOffset = (size.Z / 2) + 1.2
    local topOffset = (size.Y / 2) + 0.8

    local scanTimer = 0
    local pathTimer = 0
    local lastPathPos = selectedDrone.Position

    local orbitCenter = droneStartPosition + Vector3.new(0, 40, 0)
    local orbitRadius = 450
    local orbitAngularSpeed = 0.2

    if renderConn then renderConn:Disconnect() end
    renderConn = RunService.RenderStepped:Connect(function(dt)
        if not isDroneSteering or not selectedDrone or not selectedDrone.Parent or not selectedDrone:IsDescendantOf(Workspace) then
            hardResetCamera() return
        end

        droneTimer = droneTimer + dt

        if droneTimer >= 220 and not isDroneReturning then
            isDroneReturning = true
        end

        pathTimer = pathTimer + dt
        if pathTimer >= 0.3 then
            pathTimer = 0
            local currentPos = selectedDrone.Position
            if lastPathPos and (currentPos - lastPathPos).Magnitude > 6 then
                local linePart = Instance.new("Part")
                linePart.Name = "DronePathSegment"
                linePart.Anchored = true
                linePart.CanCollide = false
                linePart.Material = Enum.Material.Neon
                linePart.Color = Color3.fromRGB(0, 200, 255)
                linePart.Transparency = 0.2
                local dist = (currentPos - lastPathPos).Magnitude
                linePart.Size = Vector3.new(0.3, 0.3, dist)
                linePart.CFrame = CFrame.new(lastPathPos, currentPos) * CFrame.new(0, 0, -dist / 2)
                linePart.Parent = Workspace
                table.insert(dronePathParts, linePart)
                lastPathPos = currentPos
            end
        end

        local currentPos = selectedDrone.Position

        if isDroneReturning then
            local distToStart = (currentPos - droneStartPosition).Magnitude
            if distToStart < 15 then
                HUD.Visible = false
                if renderConn then renderConn:Disconnect() renderConn = nil end
                showMapSummaryGUI()
                return
            else
                local returnDir = (droneStartPosition - currentPos).Unit
                selectedDrone.Anchored = false
                pcall(function() selectedDrone:SetNetworkOwner(LocalPlayer) end)
                
                local targetCF = CFrame.new(currentPos + returnDir * droneSpeed * dt, droneStartPosition)
                selectedDrone.CFrame = selectedDrone.CFrame:Lerp(targetCF, math.clamp(dt * 6, 0, 1))
                selectedDrone.AssemblyLinearVelocity = returnDir * droneSpeed
                cam.CFrame = selectedDrone.CFrame * CFrame.new(0, topOffset, -frontOffset)
                
                BatteryLabel.Text = "DRONE: ĐANG VỀ CĂN CỨ"
                BatteryLabel.TextColor3 = Color3.fromRGB(255, 165, 0)
                TelemetryLabel.Text = string.format("RETURNING | DIST: %dM", math.floor(distToStart))
                return
            end
        end

        -- HỆ THỐNG QUÉT VÀ CHUI LỦI VÀO KHE HỎ / LỐI VÀO SIÊU NHỎ
        local desiredPos = Vector3.zero
        local currentActiveSpeed = droneSpeed

        if not isPenetratingGap and #scannedMapData.entrances > 0 then
            -- Tìm lối vào gần nhất chưa khám phá kỹ
            for _, entPos in ipairs(scannedMapData.entrances) do
                if (currentPos - entPos).Magnitude < 600 then
                    currentEntranceTarget = entPos
                    isPenetratingGap = true
                    penetrationTimer = 0
                    break
                end
            end
        end

        if isPenetratingGap and currentEntranceTarget then
            penetrationTimer = penetrationTimer + dt
            currentActiveSpeed = 90 -- Giảm tốc độ để kiểm soát tuyệt đối khi lách qua khe hẹp
            desiredPos = currentEntranceTarget

            -- Nếu đã tiến sát lối vào (hoặc xuyên qua bên trong được 4 giây), kết thúc thám hiểm kẽ hở đó và quay lại quét map
            if (currentPos - currentEntranceTarget).Magnitude < 15 or penetrationTimer > 4.5 then
                isPenetratingGap = false
                currentEntranceTarget = nil
            end
        else
            -- Quỹ đạo bay vòng tròn bao phủ toàn bộ map
            local angle = droneTimer * orbitAngularSpeed
            local targetX = orbitCenter.X + math.cos(angle) * orbitRadius
            local targetZ = orbitCenter.Z + math.sin(angle) * orbitRadius
            local targetY = orbitCenter.Y + math.sin(droneTimer * 0.35) * 60
            desiredPos = Vector3.new(targetX, targetY, targetZ)
        end

        -- Vô hiệu hóa va chạm phần cứng của drone để drone có thể chui lọt qua mọi khe hở / ống nước / tường nhỏ nhất
        for _, p in ipairs(selectedDrone:GetDescendants()) do
            if p:IsA("BasePart") then p.CanCollide = false end
        end
        selectedDrone.CanCollide = false

        -- Tự động né vật cản tĩnh
        local rayParams = RaycastParams.new()
        rayParams.FilterType = Enum.RaycastFilterType.Exclude
        rayParams.FilterDescendantsInstances = {selectedDrone, LocalPlayer.Character}
        
        local lookDir = (desiredPos - currentPos).Unit
        local rayResult = Workspace:Raycast(currentPos, lookDir * 100, rayParams)
        if rayResult and not isPenetratingGap then
            table.insert(scannedMapData.blocked, rayResult.Position)
            desiredPos = desiredPos + Vector3.new(0, 90, 0)
        end

        selectedDrone.Anchored = false
        pcall(function() selectedDrone:SetNetworkOwner(LocalPlayer) end)
        
        -- Di chuyển cực kỳ mượt mà, chính xác đến từng centimet
        local targetCF = CFrame.new(currentPos, desiredPos)
        selectedDrone.CFrame = selectedDrone.CFrame:Lerp(targetCF, math.clamp(dt * 6, 0, 1))
        selectedDrone.AssemblyLinearVelocity = selectedDrone.CFrame.LookVector * currentActiveSpeed
        
        cam.CFrame = selectedDrone.CFrame * CFrame.new(0, topOffset, -frontOffset)
        
        for _, part in ipairs(droneHiddenParts) do
            if part and part.Parent then part.LocalTransparencyModifier = 1 end
        end

        scanTimer = scanTimer + dt
        if scanTimer >= 1.2 then
            scanTimer = 0
            scanMapForSecrets(selectedDrone.Position)
        end

        local heading = math.floor((360 - math.deg(math.atan2(-selectedDrone.CFrame.LookVector.X, -selectedDrone.CFrame.LookVector.Z))) % 360)
        CompassLabel.Text = string.format("%03d° %s", heading, getCompassDirection(heading))

        local modeText = isPenetratingGap and "⚡ ĐANG CHUI QUA KHE HỎ / LỐI VÀO BÍ MẬT" else "🔄 ĐANG QUÉT VÒNG TRÒN TOÀN MAP"
        local remainingTime = math.max(0, 220 - math.floor(droneTimer))
        BatteryLabel.Text = string.format("TIME: %ds | %d ENTRANCES FOUND", remainingTime, #scannedMapData.entrances)
        BatteryLabel.TextColor3 = isPenetratingGap and Color3.fromRGB(255, 0, 0) or Color3.fromRGB(0, 180, 216)

        TelemetryLabel.Text = string.format("%s | POS: X:%d Y:%d Z:%d", modeText, math.floor(currentPos.X), math.floor(currentPos.Y), math.floor(currentPos.Z))
    end)
end

local function triggerB52Action()

    cam.CameraType = Enum.CameraType.Scriptable
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local startPos = root and root.Position or Vector3.new(0, 0, 0)
    
    local skyPos = startPos + Vector3.new(0, 3500, 0)
    cam.CFrame = CFrame.new(skyPos, skyPos + Vector3.new(0, -1, -0.01))

    if #selectedTargets > 0 then
        for _, item in ipairs(selectedTargets) do
            if item.hrp and item.hrp.Parent then 
                table.insert(rawTargets, item.hrp) 
            end
        end
    end
    
    if #rawTargets == 0 then
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                table.insert(rawTargets, p.Character.HumanoidRootPart)
            end
        end
    end

    for _, hrp in ipairs(rawTargets) do
        if hrp and hrp.Parent then
            if not isPositionBlocked(hrp.Position) then
                table.insert(targetsHRP, hrp)
            end
        end
    end
    
    if #targetsHRP == 0 then
        targetsHRP = rawTargets
    end
    
    local nameLabelsList = {}
    for _, hrp in ipairs(targetsHRP) do
        local model = hrp.Parent
        local p = Players:GetPlayerFromCharacter(model)
        local targetName = p and p.Name or model.Name
        
        if model then
            local bg = Instance.new("BillboardGui", CoreGui)
            bg.Adornee = hrp
            bg.Size = UDim2.new(0, 75, 0, 18)
            bg.StudsOffset = Vector3.new(0, 3.2, 0)
            bg.AlwaysOnTop = true

            nameLabel.Size = UDim2.new(1, 0, 1, 0)
            nameLabel.BackgroundColor3 = Color3.fromRGB(200, 0, 0)

            table.insert(nameLabelsList, nameLabel)
            
            local hl = Instance.new("Highlight", CoreGui)
            hl.Adornee = model
            hl.FillColor = Color3.fromRGB(255, 0, 0)
            hl.OutlineColor = Color3.fromRGB(255, 0, 0)
            hl.FillTransparency = 0.5
            table.insert(b52Markers, hl)
        end

    if blinkConn then blinkConn:Disconnect() end
    blinkConn = RunService.RenderStepped:Connect(function()
        local alpha = (math.sin(tick() * 16) + 1) / 2
        for _, label in ipairs(nameLabelsList) do
            if label and label.Parent then
                label.BackgroundTransparency = 0.1 + (alpha * 0.5)
            end
        end
    end)
    
    local activeBombs = {}
    if #targetsHRP > 0 and #selectedBombs > 0 then
        for i, bomb in ipairs(selectedBombs) do
            if bomb and bomb.Parent then
                local targetIndex = ((i - 1) % #targetsHRP) + 1

                    })

    b52MoveConn = RunService.Heartbeat:Connect(function(dt)

                    table.insert(dhBombs, obj)

                    local dist = (candidateHrp.Position - hrp.Position).Magnitude
                    if dist < closestDist then
                        closestDist = dist
                        closestHrp = candidateHrp
                    end

        if closestHrp then

                redBeam.Width1 = 2.0

                redBeam.LightEmission = 1
                redBeam.Transparency = NumberSequence.new(0.05)
                table.insert(dhMarkers, redBeam)
                
                table.insert(activeBombs, {
                    bomb = bomb, 
                    target = assignedTarget
                })
            end
        end
    end
    
    local dhMoveConn

            for index = #activeBombs, 1, -1 do
                local item = activeBombs[index]

                        if p:IsA("BasePart") then p.CanCollide = false end

                    local targetPos = target.Position
